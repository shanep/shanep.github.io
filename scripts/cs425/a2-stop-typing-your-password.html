Steps 0 to 4 · get there, keys, config, and what the second login saves

ssh <username>@onyx.boisestate.edu                  # step 0: password login must work before anything else
ssh-keygen -t ed25519 -C "you@boisestate.edu"       # step 1, on your laptop
ssh-copy-id <username>@onyx.boisestate.edu           # step 2, the last password you type
ssh onyx uname -n                                    # step 3, after ~/.ssh/config is in place
ssh -O exit onyx 2>/dev/null; time ssh onyx true; time ssh onyx true    # step 4, cold then warm
ping -c 3 onyx.boisestate.edu                        # step 4, the round trip
